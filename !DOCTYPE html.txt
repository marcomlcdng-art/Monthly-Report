function switchTab(type, btn) {
  document.querySelectorAll('.tab').forEach(b=>b.classList.remove('active'));
  btn.classList.add('active');
  const table = document.getElementById('compTable');
  table.querySelectorAll('tr').forEach(row => {
    Array.from(row.cells).forEach((cell, idx) => {
      if (idx === 0) { cell.style.display=''; return; }
      const colIdx = idx - 1;
      const metricIdx = Math.floor(colIdx / 3);
      const group = metrics[metricIdx]?.dataGroup;
      if (type === 'all') cell.style.display = '';
      else if (type === 'traffic') cell.style.display = group === 'traffic' ? '' : 'none';
      else cell.style.display = group === 'ads' ? '' : 'none';
    });
  });
}
